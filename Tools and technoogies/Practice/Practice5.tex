	\section{Organizing Information}
	Information is often presented as lists. \LaTeX supports both itemized and
	numbered lists.
	\subsection{Custom Itemize}
	\begin{itemize}
		\item Normal bullet
		\item[!] Urgent task
		\item[$\blacksquare$] Completed work
	\end{itemize}
	\begin{itemize}
		\item[NOTE] Special note
	\end{itemize}
	\subsection{Numbered List}
	\begin{enumerate}
		\item First numbered point
			\item[!] Custom exclamation
		\item[$\blacksquare$] Black sqaure instead of number
	\end{enumerate}
	\begin{itemize}
		\item[NOTE] Replace number with text
	\end{itemize}
	\subsection{Nested Example}
	\begin{enumerate}
		\item Outer level
		\begin{itemize}
			\item Inner bullet
			\item[Note:] This is a note
			\item[Wait!] This is a warning 
		\end{itemize}
	\end{enumerate}
